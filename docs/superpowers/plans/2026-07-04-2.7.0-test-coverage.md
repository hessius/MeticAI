**Goal:** Add the two 2.7.0 test-coverage deliverables — an end-to-end Playwright spec for the AI shot-analysis flow (#530) and self-hosted Storybook visual-regression testing to catch iOS/iPad layout regressions (#529) — without touching any production code that's under 2.6.0 TestFlight.

**Architecture:** Both deliverables are **additive test infrastructure** committed directly to `version/2.6.0` (test-only, safe during the beta). The E2E spec drives the SPA in its **default browser (proxy) transport mode** and mocks the MeticAI backend `/api/shots/*` endpoints directly with Playwright `page.route`. **This was confirmed by the A0 spike** (see Task A0 findings): a plain Playwright browser test against the Vite dev server runs in proxy mode — `getMachineMode()` returns `'proxy'` (`machineMode.ts:50-67`) — where there is **no `window.fetch` override**, so `page.route` intercepts every `/api/*` call cleanly. (The `window.fetch` override that blocks `page.route` only exists in the native/direct runtime — `main.tsx:24` installs it solely when `isDirectMode() && !isDemoMode()`, which is never true for a browser test. Demo mode is a dead end here: `ShotAnalysisView` uses raw `fetch('/api/shots/recent')` that demo mode does **not** intercept, so it returns Vite's `index.html`.) Because the backend response is mocked directly, this exercises the **UI/rendering pipeline** (shot list → detail → static facts → taste gate → LLM analysis text → parse/recommendations/error surfaces). Provider routing and the in-browser validator/repair pipeline are **direct-mode internals** already covered by unit tests (`DirectModeInterceptor.test.ts`, `OpenAICompatProvider.test.ts`, `GeminiProvider.test.ts`) and are intentionally **not** re-tested at the e2e level. Visual regression uses `@storybook/test-runner` + `jest-image-snapshot` over a device-viewport matrix, wired as an **advisory (non-blocking) CI job** mirroring `ai-guard.yml`.

**Tech Stack:** Playwright `^1.61`, `@playwright/test`, Vite dev server (`localhost:5173`), Storybook 10.4 (`@storybook/react-vite`), `@storybook/test-runner`, `jest-image-snapshot`, Bun 1.3.14, GitHub Actions.

```ts
export interface MockAnalysisOptions {
  /** llm_analysis text returned by POST /api/shots/analyze-llm (default VALID_ANALYSIS). */
  llmContent?: string
  /** If >= 400, POST /api/shots/analyze-llm fails with this status + detail message. */
  llmStatus?: number
  /** detail.message returned on llm error (surfaced verbatim by ShotDetail). */
  llmErrorMessage?: string
}

/** Route every /api/shots/* endpoint the analysis flow touches. Returns a
 *  mutable state object recording analyze-llm hit count (for re-analyze). */
export async function mockAnalysisApi(page: Page, opts: MockAnalysisOptions = {}) {
  const state = { llmCalls: 0, staticCalls: 0 }
  const json = (body: unknown, status = 200) => ({
    status,
    contentType: 'application/json',
    body: JSON.stringify(body),
  })

  await page.route('**/api/shots/recent/by-profile**', (r) => r.fulfill(json({ profiles: [] })))
  await page.route('**/api/shots/recent**', (r) => r.fulfill(json({ shots: [SHOT_SUMMARY] })))
  await page.route('**/api/shots/by-profile/**', (r) =>
    r.fulfill(json({ profile_name: SHOT_SUMMARY.profile_name, shots: [SHOT_SUMMARY], count: 1, limit: 20, is_stale: false })),
  )
  await page.route('**/api/shots/data/**', (r) => r.fulfill(json({ data: SHOT_DATA })))
  await page.route('**/api/shots/annotations**', (r) => r.fulfill(json({})))
  await page.route('**/api/shots/dates**', (r) => r.fulfill(json([])))
  await page.route('**/api/shots/llm-analysis-cache**', (r) => r.fulfill(json({ cached: false })))
  await page.route('**/api/shots/analyze', (r) => {
    state.staticCalls++
    r.fulfill(json({ status: 'success', analysis: STATIC_FACTS }))
  })
  await page.route('**/api/shots/analyze-llm', (r) => {
    state.llmCalls++
    if (opts.llmStatus && opts.llmStatus >= 400) {
      r.fulfill(json({ detail: { message: opts.llmErrorMessage ?? 'mock error' } }, opts.llmStatus))
      return
    }
    r.fulfill(json({ status: 'success', llm_analysis: opts.llmContent ?? VALID_ANALYSIS, cached: false }))
  })

  return state
}
```

```ts
// apps/web/.storybook/test-runner.ts
import type { TestRunnerConfig } from '@storybook/test-runner'
import { toMatchImageSnapshot } from 'jest-image-snapshot'

// Viewport matrix targeting the surfaces that regressed on iOS/iPad.
const VIEWPORTS: Record<string, { width: number; height: number }> = {
  iphone: { width: 390, height: 844 },      // iPhone 14
  ipadPortrait: { width: 834, height: 1194 }, // iPad Air portrait
  ipadLandscape: { width: 1194, height: 834 },
}
const VP = process.env.SB_VIEWPORT ?? 'iphone'

const config: TestRunnerConfig = {
  setup() {
    expect.extend({ toMatchImageSnapshot })
  },
  async preVisit(page) {
    await page.emulateMedia({ reducedMotion: 'reduce' })
    await page.setViewportSize(VIEWPORTS[VP])
  },
  async postVisit(page, context) {
    // Wait for fonts + a paint tick so snapshots are stable.
    await page.evaluate(() => document.fonts.ready)
    await page.waitForTimeout(150)
    const image = await page.screenshot()
    // @ts-expect-error jest-image-snapshot augments expect at runtime.
    expect(image).toMatchImageSnapshot({
      customSnapshotsDir: `__image_snapshots__/${VP}`,
      customSnapshotIdentifier: context.id,
      failureThreshold: 0.02,
      failureThresholdType: 'percent',
    })
  },
}
export default config
```
