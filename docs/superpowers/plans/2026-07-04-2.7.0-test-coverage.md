```ts
/** Seed localStorage BEFORE app scripts run: demo mode + a hosted
 *  OpenAI-compatible provider so the analyze-llm path calls /chat/completions. */
export async function seedDemoAi(
  page: Page,
  opts: { provider?: 'openai' | 'deepseek' | 'gemini'; model?: string } = {},
): Promise<void> {
  const provider = opts.provider ?? 'openai'
  const model = opts.model ?? 'gpt-test'
  await page.addInitScript(
    ([provider, model]) => {
      localStorage.setItem('meticai-machine-url', 'demo')
      localStorage.setItem('meticai-ai-mode', 'hosted')
      localStorage.setItem('meticai-ai-provider', provider)
      if (provider === 'gemini') {
        localStorage.setItem('meticai-gemini-key', 'test-key')
        localStorage.setItem('meticai-gemini-model', model)
      } else {
        localStorage.setItem(`meticai-ai-key-${provider}`, 'test-key')
        localStorage.setItem(`meticai-ai-model-${provider}`, model)
      }
    },
    [provider, model] as const,
  )
}

/** Intercept the OpenAI-compatible /chat/completions call and return canned
 *  content. Records how many times it was hit (for re-analyze assertions). */
export function mockOpenAICompat(
  page: Page,
  opts: { content?: string; status?: number; hostGlob?: string } = {},
) {
  const state = { calls: 0 }
  const glob = opts.hostGlob ?? '**/chat/completions'
  page.route(glob, async (route) => {
    state.calls++
    if (opts.status && opts.status >= 400) {
      await route.fulfill({ status: opts.status, contentType: 'application/json', body: '{"error":{"message":"mock"}}' })
      return
    }
    await route.fulfill({
      status: 200,
      contentType: 'application/json',
      body: JSON.stringify({ choices: [{ message: { content: opts.content ?? '' } }] }),
    })
  })
  return state
}

/** Intercept the Gemini SDK network call. */
export function mockGemini(page: Page, opts: { content?: string; status?: number } = {}) {
  const state = { calls: 0 }
  page.route('**/generativelanguage.googleapis.com/**', async (route) => {
    state.calls++
    if (opts.status && opts.status >= 400) {
      await route.fulfill({ status: opts.status, contentType: 'application/json', body: '{"error":{"code":' + opts.status + ',"message":"mock"}}' })
      return
    }
    await route.fulfill({
      status: 200,
      contentType: 'application/json',
      body: JSON.stringify({ candidates: [{ content: { parts: [{ text: opts.content ?? '' }] } }] }),
    })
  })
  return state
}
```

```ts
// apps/web/.storybook/test-runner.ts
import type { TestRunnerConfig } from '@storybook/test-runner'
import { toMatchImageSnapshot } from 'jest-image-snapshot'

// Viewport matrix targeting the surfaces that regressed on iOS/iPad.
const VIEWPORTS: Record<string, { width: number; height: number }> = {
  iphone: { width: 390, height: 844 },      // iPhone 14
  ipadPortrait: { width: 834, height: 1194 }, // iPad Air portrait
  ipadLandscape: { width: 1194, height: 834 },
}
const VP = process.env.SB_VIEWPORT ?? 'iphone'

const config: TestRunnerConfig = {
  setup() {
    expect.extend({ toMatchImageSnapshot })
  },
  async preVisit(page) {
    await page.emulateMedia({ reducedMotion: 'reduce' })
    await page.setViewportSize(VIEWPORTS[VP])
  },
  async postVisit(page, context) {
    // Wait for fonts + a paint tick so snapshots are stable.
    await page.evaluate(() => document.fonts.ready)
    await page.waitForTimeout(150)
    const image = await page.screenshot()
    // @ts-expect-error jest-image-snapshot augments expect at runtime.
    expect(image).toMatchImageSnapshot({
      customSnapshotsDir: `__image_snapshots__/${VP}`,
      customSnapshotIdentifier: context.id,
      failureThreshold: 0.02,
      failureThresholdType: 'percent',
    })
  },
}
export default config
```
